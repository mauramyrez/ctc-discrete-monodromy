\subsection{Clasificación eléctrica de los materiales}

\subsubsection{Conductores (modelo de la parte interna)}

\begin{cajateoria}[Átomo de cobre --- idea clave]
\begin{itemize}[leftmargin=1.1em]
  \item Núcleo: 29 protones. Orbitales: $2+8+18+1$.
  \item \textbf{Parte interna} $=$ núcleo $+$ orbitales internos $= (+29)+(-28)=+1$.
  \item El electrón de valencia (orbital exterior) se atrae débilmente $\Rightarrow$ \textbf{electrón libre}.
  \item Incluso una tensión pequeña desplaza electrones libres $\Rightarrow$ buen conductor.
\end{itemize}
\end{cajateoria}

\subsubsection{Aislantes}
Átomos con 5--7 electrones de valencia (óptimo: \textbf{ocho}). El costo energético de completar el octeto es menor que el de liberar electrones $\Rightarrow$ no hay movilidad de carga.

\subsubsection{Semiconductores: Si y Ge}
\begin{itemize}[leftmargin=1.1em]
  \item Mejores semiconductores: \textbf{4 electrones de valencia}.
  \item \textbf{Ge}: históricamente primero; desventaja: corriente inversa excesiva.
  \item \textbf{Si}: más abundante tras el O$_2$; base de la electrónica moderna.
  \item Si aislado: 14 protones, 14 electrones; orbitales $2+8+4$. Parte interna: $+4$.
\end{itemize}

\subsection{Construcción de semiconductores}

\begin{cajanota}[Dos clases de materiales]
\begin{itemize}[leftmargin=1.1em]
  \item \textbf{Monocristalinos:} Ge, Si (estructura cristalina repetitiva).
  \item \textbf{Compuestos:} GaAs, CdS, GaN, GaAsP.
  \item Los más usados en dispositivos: \textbf{Ge, Si, GaAs}.
\end{itemize}
\end{cajanota}

\subsubsection{Redes cristalinas y enlace covalente}

Cada átomo de Si comparte 1 electrón con \textbf{4 vecinos} $\Rightarrow$ 8 electrones en el orbital de valencia.

\begin{cajaformula}[Saturación de valencia]
\begin{equation}
n = 8 \eqtag{1.1}
\end{equation}
El orbital de valencia \textbf{no puede} contener más de ocho electrones. Esos ocho son \textbf{electrones ligados}. A temperatura ambiente, un cristal de Si puro es casi un aislante perfecto.
\end{cajaformula}

\subsubsection{Hueco, recombinación y tiempo de vida}

\begin{cajateoria}[Generación térmica de portadores]
Sobre el cero absoluto ($-273\,^\circ\mathrm{C}$), las vibraciones térmicas pueden liberar un electrón del orbital de valencia hacia un nivel superior $\Rightarrow$ se crea un par:
\begin{center}
\textbf{electrón libre} $+$ \textbf{hueco} (vacante con comportamiento de carga positiva).
\end{center}
\end{cajateoria}

\begin{center}
\begin{tikzpicture}[
  every node/.style={font=\small},
  proceso/.style={draw,rounded corners,align=center,minimum width=2.8cm,minimum height=1.1cm}
]
\node[proceso,fill=orange!20] (g) {Energía térmica\\$\rightarrow$ generación};
\node[proceso,fill=yellow!25,right=1.2cm of g] (p) {Par $e^-$ libre\\$+$ hueco};
\node[proceso,fill=green!15,right=1.2cm of p] (r) {Recombinación\\$e^-$ cae en hueco};
\draw[-{Latex},thick] (g) -- (p);
\draw[-{Latex},thick] (p) -- (r);
\draw[-{Latex},thick,dashed] (r.south) -- ++(0,-0.5) -| node[below,pos=0.25,font=\scriptsize]{ciclo continuo} (g.south);
\end{tikzpicture}
\end{center}

\begin{itemize}[leftmargin=1.1em]
  \item \textbf{Recombinación:} unión de un $e^-$ libre con un hueco.
  \item \textbf{Tiempo de vida:} intervalo entre creación y desaparición del $e^-$ libre (ns a $\mu$s).
  \item En cualquier instante, en Si puro: (1)~generación, (2)~recombinación, (3)~portadores temporales pendientes de recombinar.
\end{itemize}

\subsection{Semiconductores intrínsecos}

\begin{cajanota}[Definición]
Semiconductor \textbf{intrínseco} $=$ cristal puro (cada átomo es Si). A $T$ ambiente actúa casi como aislante: pocos $e^-$ libres y huecos (pares térmicos).
\end{cajanota}

\end{cajaformula}

\subsection{Semiconductores extrínsecos (dopado)}

\begin{cajateoria}[Dopado]
Añadir impurezas ($\sim 1$ parte en $10^7$) a un cristal intrínseco para alterar la conductividad. El material dopado es \textbf{extrínseco}.
\begin{itemize}[leftmargin=1.1em]
  \item Débilmente dopado $\Rightarrow$ $R$ alta.
  \item Fuertemente dopado $\Rightarrow$ $R$ baja.
\end{itemize}
\end{cajateoria}

\section{Teoría de circuitos --- conceptos básicos}

\begin{cajaalerta}[Convención de signos (crítica)]
Prestar atención a los signos ``$+$''/``$-$'' de tensión y a la flecha de corriente: suelen decidir si la respuesta es correcta o incorrecta.
\end{cajaalerta}

\subsection{Carga, corriente, potencial y potencia}

\subsubsection{Carga eléctrica $q$}
\begin{itemize}[leftmargin=1.1em]
  \item Unidad: \textbf{coulomb} (C).
  \item $1\,\mathrm{C}$ contiene $1/(1.602\times 10^{-19})=6.24\times 10^{18}$ electrones.
  \item En laboratorio: pC, nC, $\mu$C.
  \item Carga elemental: $e=-1.602\times 10^{-19}\,\mathrm{C}$ (solo múltiplos enteros en la naturaleza).
  \item \textbf{Conservación de la carga:} no se crea ni destruye, solo se transfiere.
\end{itemize}

\begin{cajaformula}[Ejemplo tipo rayo]
Flujo máximo $\approx 20\,000\,\mathrm{C/s}$ durante $100\,\mu\mathrm{s}$:
\[
Q=(20\,000\,\mathrm{C/s})(100\times 10^{-6}\,\mathrm{s})=2\,\mathrm{C},\qquad
N_e=\frac{Q}{|q_e|}=\frac{2}{1.602\times 10^{-19}}=1.248\times 10^{19}\ \text{electrones.}
\]
\end{cajaformula}

\subsubsection{Corriente $i$}
\begin{cajaformula}[Definición]
\begin{equation}
i=\frac{\mathrm{d}q}{\mathrm{d}t} \eqtag{2.1}
\end{equation}
$1\,\mathrm{A}=1\,\mathrm{C/s}$. Integrando:
\begin{equation}
q=\int_{t_0}^{t} i\,\mathrm{d}t \eqtag{2.2}
\end{equation}
\end{cajaformula}

\begin{itemize}[leftmargin=1.1em]
  \item \textbf{CD (DC):} constante $\rightarrow$ símbolo $I$.
  \item \textbf{CA (AC):} variable (p.\,ej.\ senoidal) $\rightarrow$ símbolo $i$.
  \item Convención: sentido de $i$ $=$ movimiento de \textbf{carga positiva}.
\end{itemize}

\subsubsection{Potencial / tensión $v$}
Trabajo por unidad de carga. Unidad: \textbf{volt} (V); $1\,\mathrm{V}=1\,\mathrm{J/C}$.
\begin{equation}
V=\frac{W}{Q} \eqtag{2.3}
\end{equation}
Puede existir tensión \textbf{sin} corriente. Convención: si $i>0$ entra por A y una fuente externa gasta energía, A es positiva respecto a B.

\subsubsection{Potencia $p$}
\begin{cajaformula}[Potencia absorbida]
\begin{equation}
p=vi \eqtag{2.4}
\end{equation}
Si la corriente entra por el terminal ``$+$'': $p=vi$ es potencia \textbf{absorbida}. Si $p<0$, el elemento \textbf{genera} potencia.
\end{cajaformula}

\subsection{Capacitores}

\begin{cajateoria}[Definición]
Elemento pasivo que almacena energía en su campo eléctrico: dos placas conductoras separadas por dieléctrico.
\begin{equation}
q=Cv \eqtag{2.5}
\end{equation}
$C$ en \textbf{faradios} (F); $1\,\mathrm{F}=1\,\mathrm{C/V}$.
\end{cajateoria}

\begin{cajaformula}[Placas paralelas y $i$--$v$]
\begin{equation}
C=\varepsilon\frac{A}{d} \eqtag{2.6}
\end{equation}
\begin{align}
i&=C\frac{\mathrm{d}v}{\mathrm{d}t} \eqtag{2.7}\\
v&=\frac{1}{C}\int_{-\infty}^{t} i\,\mathrm{d}t
=\frac{1}{C}\int_{t_0}^{t} i\,\mathrm{d}t+v(t_0) \eqtag{2.8--2.9}
\end{align}
Factores: $\uparrow A\Rightarrow\uparrow C$; $\downarrow d\Rightarrow\uparrow C$; $\uparrow\varepsilon\Rightarrow\uparrow C$.
\end{cajaformula}

\begin{cajanota}[Propiedades del capacitor]
\begin{enumerate}[leftmargin=1.2em]
  \item En CD ($v$ constante): $i=0$ $\Rightarrow$ \textbf{circuito abierto} en CD.
  \item $v_C$ es continua: \textbf{no} puede cambiar abruptamente.
  \item Ideal: no disipa energía (almacena / devuelve).
  \item Real: resistencia de fuga en paralelo (hasta $\sim 100\,\mathrm{M}\Omega$; suele despreciarse).
\end{enumerate}
\end{cajanota}

\begin{cajaformula}[Capacitores equivalentes]
\textbf{Paralelo} (como $R$ en serie):
\begin{equation}
C_{\mathrm{eq}}=C_1+C_2+\cdots+C_N \eqtag{2.10}
\end{equation}
\textbf{Serie} (como $R$ en paralelo):
\begin{equation}
\frac{1}{C_{\mathrm{eq}}}=\frac{1}{C_1}+\frac{1}{C_2}+\cdots+\frac{1}{C_N} \eqtag{2.11}
\end{equation}
\end{cajaformula}

\subsection{Impedancia}

En dominio fasorial / frecuencia:
\begin{equation}
\mathbf{V}=R\,\mathbf{I},\qquad
\mathbf{V}=j\omega L\,\mathbf{I},\qquad
\mathbf{V}=\frac{1}{j\omega C}\,\mathbf{I} \eqtag{2.12--2.13}
\end{equation}

\begin{cajaformula}[Impedancia $Z$]
\[
Z=\frac{\mathbf{V}}{\mathbf{I}}\quad [\ohm]
\]
\begin{itemize}[leftmargin=1.1em]
  \item $Z$ es compleja; \textbf{no} es un fasor ni se transforma al tiempo con $e^{j\omega t}$.
  \item Forma rectangular: $Z=R+jX$ ($X=$ \textbf{reactancia}).
  \item Forma polar: $Z=|Z|\angle\theta$.
  \item $R$ ideal: $X=0$. $L$ o $C$ ideales: puramente reactivos.
  \item Serie/paralelo de $Z$: mismas reglas que para resistencias.
\end{itemize}
Impedancias elementales:
\[
Z_R=R,\qquad Z_L=j\omega L,\qquad Z_C=\frac{1}{j\omega C}=-j\frac{1}{\omega C}.
\]
\end{cajaformula}

\subsection{Ley de Ohm y potencia en resistores}

\begin{cajaformula}[Ley de Ohm]
\begin{equation}
v=Ri \eqtag{2.14}
\end{equation}
Resistencia lineal: recta $i$--$v$ por el origen (modelo idealizado).
Potencia disipada:
\begin{equation}
p=vi=i^2 R=\frac{v^2}{R} \eqtag{2.15}
\end{equation}
Como $R>0$, $p>0$ siempre: el resistor \textbf{siempre absorbe} potencia (pasivo).
\end{cajaformula}

\subsection{Vocabulario de redes}
\begin{itemize}[leftmargin=1.1em]
  \item \textbf{Nodo:} punto común a dos o más elementos.
  \item \textbf{Trayectoria:} secuencia de nodos/elementos sin repetir nodo.
  \item \textbf{Lazo:} trayectoria cerrada.
  \item \textbf{Rama:} un elemento simple con sus dos nodos.
\end{itemize}

\subsection{Ley de corrientes de Kirchhoff (LCK)}
\begin{cajaformula}[LCK]
La suma algebraica de las corrientes que entran a cualquier nodo es cero:
\begin{equation}
\sum_{n=1}^{N} i_n = 0 \eqtag{2.18--2.19}
\end{equation}
Equivalente: $\sum i_{\mathrm{entran}}=\sum i_{\mathrm{salen}}$. (La carga no se acumula en un nodo.)
\end{cajaformula}

\subsection{Ley de voltajes de Kirchhoff (LVK)}
\begin{cajaformula}[LVK]
La suma algebraica de las tensiones alrededor de cualquier trayectoria cerrada es cero:
\begin{equation}
\sum_{n=1}^{N} v_n = 0 \eqtag{2.20--2.21}
\end{equation}
Método práctico: recorrer el lazo en sentido horario; escribir $+v$ al entrar por ``$+$'' y $-v$ al entrar por ``$-$''.
\end{cajaformula}

\subsection{Divisor de tensión y de corriente}

\begin{equation}
v_k = v\cdot\frac{R_k}{R_1+R_2+\cdots+R_N} \eqtag{2.22}
\end{equation}
Para dos resistencias:
\[
v_1=v\frac{R_1}{R_1+R_2},\qquad v_2=v\frac{R_2}{R_1+R_2}.
\]
\end{cajaformula}

\begin{equation}
i_2=i\frac{R_1}{R_1+R_2},\qquad
i_1=i\frac{R_2}{R_1+R_2} \eqtag{2.23--2.24}
\end{equation}
General ($N$ en paralelo):
\begin{equation}
i_k=i\cdot\frac{1/R_k}{1/R_1+\cdots+1/R_N} \eqtag{2.25}
\end{equation}
\end{cajaformula}

\begin{cajateoria}[Enunciado]
En una red \textbf{lineal resistiva}, la tensión o corriente en cualquier elemento es la suma algebraica de los aportes de cada fuente independiente actuando sola:
\begin{itemize}[leftmargin=1.1em]
  \item Fuentes de tensión independientes inactivas $\rightarrow$ \textbf{cortocircuito}.
  \item Fuentes de corriente independientes inactivas $\rightarrow$ \textbf{circuito abierto}.
  \item Fuentes \textbf{dependientes}: suelen permanecer activas en cada experimento.
\end{itemize}
Con $N$ fuentes independientes $\Rightarrow$ $N$ circuitos auxiliares.
\end{cajateoria}

\section{Teoremas de Thévenin y Norton}

\subsection{Thévenin}
\begin{cajaformula}[Teorema de Thévenin]
Todo circuito lineal de dos terminales $\equiv$ fuente $V_{\mathrm{Th}}$ en serie con $R_{\mathrm{Th}}$:
\begin{itemize}[leftmargin=1.1em]
  \item $V_{\mathrm{Th}}=V_{\mathrm{oc}}$: tensión en vacío en las terminales.
  \item $R_{\mathrm{Th}}$: resistencia equivalente vista desde las terminales con fuentes independientes apagadas (dependientes intactas).
\end{itemize}
\end{cajaformula}

\textbf{Procedimiento:}
\begin{enumerate}[leftmargin=1.2em]
  \item Separar red A (a simplificar) y red B (carga).
  \item Desconectar B; medir/calcular $v_{\mathrm{oc}}$ en A.
  \item Apagar fuentes indep.\ de A $\rightarrow$ red inactiva; obtener $R_{\mathrm{Th}}$.
  \item Sustituir A por $V_{\mathrm{oc}}$ en serie con $R_{\mathrm{Th}}$; reconectar B.
\end{enumerate}

\subsection{Norton}
\begin{cajaformula}[Teorema de Norton]
Circuito lineal de dos terminales $\equiv$ fuente $I_N$ en paralelo con $R_N$:
\begin{itemize}[leftmargin=1.1em]
  \item $I_N=I_{\mathrm{sc}}$: corriente de cortocircuito en las terminales.
  \item $R_N$: misma resistencia equivalente con fuentes indep.\ apagadas.
\end{itemize}
\end{cajaformula}

\subsection{Equivalencia Thévenin $\leftrightarrow$ Norton}

\begin{cajaformula}[Conversiones]
\begin{align}
R_N &= R_{\mathrm{Th}} \eqtag{2.26}\\
I_N &= \frac{V_{\mathrm{Th}}}{R_{\mathrm{Th}}} \eqtag{2.27}\\
V_{\mathrm{Th}} &= I_N\,R_N \eqtag{2.28}
\end{align}
$R_{\mathrm{Th}}$ va en \textbf{serie} con la fuente de tensión; $R_N$ va en \textbf{paralelo} con la fuente de corriente.
\end{cajaformula}
